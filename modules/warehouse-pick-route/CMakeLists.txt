cmake_minimum_required(VERSION 3.20)
project(warehouse_pick_route LANGUAGES CXX)
add_executable(pick-route src/main.cpp)
target_compile_features(pick-route PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(pick-route PRIVATE /W4 /WX)
else()
  target_compile_options(pick-route PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME route-cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_cli.py" "$<TARGET_FILE:pick-route>")
endif()
